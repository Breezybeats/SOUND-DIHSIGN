cmake_minimum_required(VERSION 3.16)
project(SoundDihsignCore VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ZLIB REQUIRED)
add_library(SoundCore STATIC Sources/SoundCore/SoundCore.cpp Sources/SoundCore/SerumPreset.cpp)
target_include_directories(SoundCore PUBLIC Sources/SoundCore/include)
target_link_libraries(SoundCore PUBLIC ZLIB::ZLIB)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(SoundCore PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(SOUND_CORE_BUILD_TESTS "Build portable audio verification" ON)
if(SOUND_CORE_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(soundcore_tests tests/SoundCoreTests.cpp)
    target_link_libraries(soundcore_tests PRIVATE SoundCore Threads::Threads)
    add_test(NAME soundcore_audio COMMAND soundcore_tests)
    add_executable(serum_preset_tests tests/SerumPresetTests.cpp)
    target_link_libraries(serum_preset_tests PRIVATE SoundCore)
    add_test(NAME serum_preset_format COMMAND serum_preset_tests)
endif()
